================================================================
🔊 ALICE_SON - ACCÈS RAPIDE
================================================================

URL D'ACCÈS :
    http://<IP_DU_PI>:8080

TROUVER L'IP DU PI :
    hostname -I

COMMANDES UTILES :
    
    Voir statut :         sudo systemctl status alice_son.service
    Redémarrer :          sudo systemctl restart alice_son.service
    Voir logs :           sudo journalctl -u alice_son.service -f
    Arrêter :             sudo systemctl stop alice_son.service
    Désactiver :          sudo systemctl disable alice_son.service
    Activer :             sudo systemctl enable alice_son.service

INSTALLER ALICE_SON :
    cd /home/alice/ALICE_PI_INSTALL
    chmod +x install_alice_son.sh
    sudo ./install_alice_son.sh

VÉRIFIER CONFIG :
    cat /home/alice/media/config.json

TESTER MANUELLEMENT :
    cd /home/alice
    python3 ALICE_SON.py

DÉPANNAGE :
    
    1. Flask non installé ?
       sudo pip3 install Flask==3.0.0 Werkzeug==3.0.1
    
    2. Port déjà utilisé ?
       sudo netstat -tulpn | grep 8080
    
    3. Permissions sudo ?
       sudo -l -U alice
    
    4. Service ne démarre pas ?
       sudo journalctl -u alice_son.service -n 50
